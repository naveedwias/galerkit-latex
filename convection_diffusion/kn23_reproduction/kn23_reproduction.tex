\begin{abstract}
P.~Knobloch, \emph{An algebraically stabilized method for convection--diffusion--reaction problems with optimal
experimental convergence rates on general meshes}, Numer.\ Algorithms 94 (2023)~\cite{Kn23}, analyses why algebraic
stabilizations lose accuracy on meshes without local symmetry and proposes the Symmetrized Monotone Upwind-type
Algebraically Stabilized (SMUAS) method. We check that galerkit implements all four methods of the paper (the AFC
schemes with the Kuzmin and the BJK limiter, MUAS and SMUAS) together with the limiter variants used in its analysis,
state the SMUAS method as an algorithm, and recompute the paper's numerical results: Tables~1--9, the statements of
Sections~5 and~6, and Figures~3, 4, 5, 7, 8, 9, 11 and~13. \input{tables/kn23_abstract.tex}
\end{abstract}

\section{The SMUAS algorithm}\label{sec:alg}
%=====================================================================================================

\paragraph{Setting.} $P_1$ elements on a simplicial triangulation of $\Omega\subset\R^d$, nodes $x_1,\dots,x_N$ with
$x_{M+1},\dots,x_N$ on $\partial\Omega$, $S_i$ the neighbours of $x_i$, $\Delta_i$ the patch of $x_i$, and
$A=(a_{ij})$ the Galerkin matrix of $-\eps\Delta u+\bb\cdot\nabla u+cu=g$, $a_{ij}=a(\varphi_j,\varphi_i)$. The method
solves
\begin{equation}\label{eq:alg}
  \sum_{j=1}^N\bigl(a_{ij}+b_{ij}(U)\bigr)u_j=g_i,\quad i\le M,\qquad u_i=u_b(x_i),\quad i>M,
\end{equation}
with the symmetric, row-sum-free stabilization matrix
$b_{ij}(U)=-\max\{\beta_{ij}(U)a_{ij},0,\beta_{ji}(U)a_{ji}\}$ ($i\ne j$), $b_{ii}=-\sum_{j\ne i}b_{ij}$.

\begin{framed}
\noindent\textbf{Algorithm SMUAS.}\quad\emph{Input:} $A$, $g$, $u_b$, the mesh, the choice (67) or (68) of the weights,
a tolerance $\tau$. \emph{Output:} $U$ solving \eqref{eq:alg}.

\smallskip\noindent\textbf{Set-up} (once per mesh).
\begin{enumerate}[label=S\arabic*., leftmargin=*, itemsep=2pt]
\item For every interior node $i$ and every $j\in S_i$: find the simplex $T_{ij}\subset\Delta_i$ entered by the half
  line $\{x_i+t(x_i-x_j):t>0\}$ (the simplex whose barycentric gradients satisfy
  $\nabla\lambda_k\cdot(x_i-x_j)\ge0$ for all its vertices $x_k\ne x_i$). Store the row
  $w_{ij}=e_i+\sum_{x_k\in T_{ij}}(\nabla\lambda_k\cdot(x_i-x_j))\,e_k$, so that the reflected value is
  $u_{ij}=w_{ij}\cdot U=u_i+\nabla u_h|_{T_{ij}}\cdot(x_i-x_j)$ \hfill(60)
\item Weights, (67): $p_{ij}=\max\{a_{ij},0,a_{ji}\}$, $q_{ij}=\max\{|a_{ij}|,a_{ji}\}$; or (68): $p_{ij}=q_{ij}=1$.
  Set $p_{ij}=0$ unless $a_{ij}>0$ or $a_{ji}>0$.
\end{enumerate}

\noindent\textbf{Stabilization matrix} $B(U)$ for a given $U$.
\begin{enumerate}[label=B\arabic*., leftmargin=*, itemsep=2pt]
\item For $i\le M$ and $j\in S_i$: $\delta_{ij}=u_i-u_j$ and $r_{ij}=u_i-u_{ij}$.
\item $P_i^\pm=\sum_{j\in S_i}p_{ij}\{(\delta_{ij})^\pm+(r_{ij})^\pm\}$,\quad
  $Q_i^\pm=\sum_{j\in S_i}q_{ij}\{(-\delta_{ij})^\pm+(-r_{ij})^\pm\}$ \hfill(61)--(64)\\
  with $z^+=\max\{0,z\}$, $z^-=\min\{0,z\}$.
\item $R_i^\pm=\min\{1,Q_i^\pm/P_i^\pm\}$, and $R_i^\pm=1$ if $P_i^\pm=0$ or $i>M$. \hfill(20), (21)
\item $\beta_{ij}=1-R_i^+$ if $u_i>u_j$, $\beta_{ij}=1-R_i^-$ if $u_i<u_j$, $\beta_{ij}=0$ if $u_i=u_j$. \hfill(36)
\item $b_{ij}=-\max\{\beta_{ij}a_{ij},0,\beta_{ji}a_{ji}\}$ for every edge, $b_{ii}=-\sum_{j\neq i}b_{ij}$. \hfill(31), (32)
\end{enumerate}

\noindent\textbf{Nonlinear solve.} Let $D$ be $B$ with all $\beta_{ij}=1$, i.e.\ $d_{ij}=-\max\{a_{ij},0,a_{ji}\}$.
\begin{enumerate}[label=N\arabic*., leftmargin=*, itemsep=2pt]
\item $U^0$: solve $(A+D)U^0=g$ (with the Dirichlet values), the monotone low-order solution.
\item For $k=0,1,\dots$: compute $B(U^k)$ by B1--B5 and the residual
  $r^k=g-(A+B(U^k))U^k$ on the free nodes; stop if $\|r^k\|_2\le\tau$.
\item \emph{Fixed point rhs} (used in~\cite{Kn23}, Remark~5): solve $(A+D)\tilde U=g-(B(U^k)-D)U^k$ with the factorization
  of $A+D$ from N1, and set $U^{k+1}=U^k+\omega_k(\tilde U-U^k)$ with a damping $\omega_k\in(0,1]$; or
  \emph{semismooth Newton}: $U^{k+1}=U^k-\lambda_k J(U^k)^{-1}r^k$ with the generalized Jacobian of the piecewise
  smooth map $U\mapsto(A+B(U))U$ and an Armijo step $\lambda_k$.
\end{enumerate}
\end{framed}

\paragraph{Properties} (proved in~\cite{Kn23} for arbitrary simplicial meshes). (A1) and hence existence of a
solution (Theorem~4), the local DMP via (A2) (Theorem~5), and linearity preservation (Remark~2: for $u_h\in P_1(\R^d)$ one has $r_{ij}=-\delta_{ij}$, hence
$P_i^+=-P_i^-=\sum_j p_{ij}|u_i-u_j|\le\sum_j q_{ij}|u_i-u_j|=Q_i^+=-Q_i^-$ by $p_{ij}\le q_{ij}$, so $R_i^\pm=1$ and
$B(U)=0$).
The cost per evaluation of $B(U)$ is one sparse product $WU$ (the reflected values) plus edge loops, i.e.\ linear in
the number of edges; the set-up S1 is a single pass over the patches.

\paragraph{Implementation in galerkit.}
\begin{itemize}[leftmargin=*, itemsep=1pt]
\item S1: \texttt{reflection\_operator}, a sparse matrix $W$ with one row per directed edge;
\item B1--B5: \texttt{AFCOperator.b\_edges};
\item N1--N3: \texttt{solve\_afc} with \texttt{afc\_solver} set to \texttt{"fp\_rhs"}, \texttt{"fp\_anderson"}
  (Anderson acceleration) or \texttt{"newton"} (exact generalized Jacobian of SMUAS, \texttt{\_jacobian\_muas}).
\end{itemize}

\section{Set-up of the computations}\label{sec:setup}
%=====================================================================================================

\begin{itemize}[leftmargin=*]
\item $P_1$ elements, no lumping of the reaction term (in contrast to~\cite{BJK16}), $\varepsilon$ as stated.
\item $ne$ is the number of edges on the part of $\partial\Omega$ on $y=1$: Grids 1, 4--8 have $ne\times ne$ squares;
  Grid~2 has $ne=2$ and Grid~3 $ne=4$ on level~0 and is refined uniformly, so that grids with the same $ne$ have
  similar numbers of triangles; Grid~9 is built for each $ne$.
\item Error norm $\|v\|_h^2=\varepsilon|v|_1^2+\sigma_0\|v\|_0^2+b_h(u_h;v,v)$ with $v=u-u_h$, the last term
  evaluated at $i_hu-u_h$, and $\sigma_0=1$. For Example~2, where $c=0$, the paper's Table~3 is reproduced only with
  $\sigma_0=1$, i.e.\ the $L^2$ term is kept although $\sigma_0=0$ there: $\sqrt{\varepsilon|e|_1^2+b_h}=8.560\cdot10^{-3}$
  against the printed $1.179\cdot10^{-2}=\sqrt{(8.560\cdot10^{-3})^2+\|e\|_0^2}$ for $ne=16$.
\item Nonlinear solver: semismooth Newton with the exact generalized Jacobian (Kuzmin, MUAS, SMUAS) or with coloured
  finite differences (the variants), stopped at a Euclidean residual of $10^{-10}$; where Newton is slow, the
  Anderson-accelerated fixed point rhs. The paper used fixed point rhs; the values do not depend on the solver once
  the residual is small.
\item ``Odd horizontal grid lines'' in (47), (48) are counted from $y=0$ as line~1, i.e.\ $y=0,2h,4h,\dots$: with
  $\mu$ on these lines Example~2 is solved exactly, with $\mu$ on the other lines it is not (error $2.5\cdot10^{-2}$).
\end{itemize}

\section{Discussion}\label{sec:disc}
%=====================================================================================================

\paragraph{Agreement.} Table~S shows the agreement for every recomputed table; together with the checks of
Section~\ref{sec:checks} and the figures, every numerical statement of~\cite{Kn23} that does not depend on the
FVCA5 meshes is reproduced (Table~5 up to $ne=256$). Three points of the set-up had to be inferred and are now fixed by the data: the
$\sigma_0=1$ in $\|\cdot\|_h$ also for $c=0$ (Table~3), the counting of the ``odd horizontal grid lines'' in
(47), (48), and the use of the (exact, recovered) Grids~2 and~3 of~\cite{BJK16}.

\paragraph{One nuance.} For Example~3 on Grid~4 the paper states that SMUAS is nodally exact except on the last
interior vertical grid line. This holds for the weights (68). With (67) the computed solution (residual
$4\cdot10^{-11}$) also has an error of about $10^{-3}$ on the second-last line; the paper does not say which weights it
used for this statement, and its remark that (67) and (68) give ``similar'' results remains true in the norms.

\paragraph{Table 10 not recomputed.} Table~10 uses a triangulated version of the \texttt{mesh4\_1} family of the
FVCA5 benchmark. The meshes are distributed only by the benchmark web site, which this environment cannot reach,
and only the coarsest of the six meshes is drawn in the paper (Fig.~12, right); the finer meshes cannot be
inferred from it. The run script reads any mesh as a \texttt{SimplexMesh}, so Table~10 can be added as soon as the
files are available.

\paragraph{Solvers.} Semismooth Newton (exact generalized Jacobian for Kuzmin, MUAS and SMUAS) converged in
7--53 iterations for Tables~1, 4 and 6--9 up to $ne=512$ (up to 40 minutes for the $5.3\cdot10^5$ unknowns of
Grid~2). For the Kuzmin limiter on Grid~4 (Tables~2 and~3) the Anderson-accelerated fixed point rhs was used at
$ne=256$ and~$512$ (8\,775 and 550 iterations at $ne=512$), as Newton becomes slow there. Table~5 (SMUAS with (68) on
Grid~4) is reported up to $ne=256$: at $ne=512$ neither Newton nor the Anderson iteration had converged after
1.5~hours, and the run was stopped. The paper's Remark~5 (fixed point rhs behaves for SMUAS as for the Kuzmin
limiter) is therefore confirmed for (67) but not tested at the finest level for (68).

\paragraph{Minimality and the sharper track.} The SMUAS properties used here are exactly those of Table~11 of the
paper: DMP on arbitrary simplicial meshes (via (A2), which needs only $p_{ij}>0$ for $a_{ij}>0$, (66)), linearity
preservation (which needs only $p_{ij}\le q_{ij}$, (65)), upwind character (the $\max\{\beta_{ij}a_{ij},0,\beta_{ji}a_{ji}\}$
form (31)), and the observed optimal rates. The last property is experimental: no a~priori estimate better than
the $O(h^{1/2})$ of (16) is proved for SMUAS, and the reproduced Tables~4--9 are the evidence. A proof of
optimal-order convergence on general meshes (the open problem stated at the end of~\cite{Kn23}) would be the
sharper result; the linearity-preserving BJK limiter~\cite{BJK17} has the DMP and linearity preservation as well but
not the upwind character, and its nonlinear problems are much harder to solve.
